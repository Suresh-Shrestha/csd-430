<%@ page contentType="text/html; charset=UTF-8" %>
<%@ taglib prefix="my"
    uri="http://example.com/mytags" %>

<!DOCTYPE html>
<html>
<head>
    <title>Custom Tag Example</title>
</head>
<body>

    <h2>My First Custom Tag</h2>

    <my:welcome />

</body>
</html>
